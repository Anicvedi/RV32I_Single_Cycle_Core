\documentclass[border=15pt,tikz]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,shapes.geometric,calc}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily\footnotesize,
  >=Stealth,
  line width=0.8pt,
  % Pure academic grayscale styles
  block/.style={draw=black, fill=white, line width=0.9pt, align=center, inner sep=4pt},
  reg/.style={draw=black, fill=gray!8, line width=0.9pt, align=center, inner sep=4pt},
  mux/.style={draw=black, fill=white, line width=0.9pt, trapezium, trapezium angle=70, text centered, inner sep=2pt},
  alu/.style={draw=black, fill=white, line width=0.9pt, shape border rotate=270, text centered},
  adder/.style={draw=black, fill=gray!8, line width=0.8pt, circle, inner sep=2pt},
  ctrl/.style={draw=black, fill=gray!5, line width=0.9pt, rounded corners=6pt, align=center, inner sep=5pt},
  bus/.style={line width=0.9pt, ->},
  ctrlwire/.style={line width=0.7pt, dashed, ->},
  dot/.style={fill=black, circle, inner sep=1.2pt}
]

  % =========================================================================
  % 1. PROGRAM COUNTER & INSTRUCTION FETCH
  % =========================================================================
  \node[reg, minimum width=1.1cm, minimum height=2.4cm] (pc) at (0, 0) {\textbf{PC}};
  
  \node[block, minimum width=2.4cm, minimum height=2.8cm, right=1.6cm of pc] (imem) {
    \textbf{Instruction}\\\textbf{Memory}\\[6pt]
    \scriptsize 512 $\times$ 32-bit\\[4pt]
    \tiny (Distributed RAM)
  };
  
  % Adder: PC + 4
  \node[block, minimum width=1.4cm, minimum height=1.6cm] (add_pc4) at (1.5, 3.4) {
    \textbf{+ 4}\\[3pt]
    \scriptsize Adder
  };

  % =========================================================================
  % 2. DECODE STAGE: REGISTER FILE, IMM GEN, CONTROL UNIT
  % =========================================================================
  % Control Unit at top center
  \node[ctrl, minimum width=3.4cm, minimum height=1.6cm] (ctrl) at (6.8, 4.4) {
    \textbf{Main Control Unit}\\[2pt]
    \scriptsize (Combinational Decoder)
  };

  % Register File
  \node[block, minimum width=2.8cm, minimum height=3.6cm] (rf) at (6.8, 0) {
    \textbf{Register File}\\[6pt]
    \scriptsize 32 $\times$ 32-bit ($x0 \equiv 0$)\\[8pt]
    \raggedright
    \tiny \hspace{4pt}rs1 [19:15] \hfill rdata1 \hspace{4pt}\\[6pt]
    \tiny \hspace{4pt}rs2 [24:20] \hfill rdata2 \hspace{4pt}\\[6pt]
    \tiny \hspace{4pt}rd [11:7]\\[6pt]
    \tiny \hspace{4pt}wdata\\[6pt]
    \tiny \hspace{4pt}RegWrite
  };

  % Immediate Generator below RF
  \node[block, minimum width=2.8cm, minimum height=1.6cm] (imm) at (6.8, -3.2) {
    \textbf{Immediate Generator}\\[3pt]
    \scriptsize (I, S, B, U, J Formats)\\[2pt]
    \tiny Sign / Zero Extension
  };

  % =========================================================================
  % 3. EXECUTE STAGE: ALU, ALUSRC MUX, BRANCH EVALUATOR
  % =========================================================================
  % ALUSrc Mux
  \node[mux, rotate=90, minimum width=1.8cm, minimum height=0.7cm] (alusrc) at (9.4, -0.6) {};
  \node at (9.4, -0.2) {\tiny 0};
  \node at (9.4, -1.0) {\tiny 1};
  \node[above=0pt of alusrc, font=\tiny] {ALUSrc};

  % 32-bit ALU
  \node[block, minimum width=2.2cm, minimum height=3.0cm] (alu) at (11.5, 0) {
    \textbf{32-bit ALU}\\[8pt]
    \tiny 10 Operations\\[4pt]
    \tiny Carry Chains\\[8pt]
    \scriptsize Result\\[4pt]
    \scriptsize Zero / Cond
  };

  % Branch Target Adder
  \node[block, minimum width=1.6cm, minimum height=1.6cm] (add_branch) at (11.5, 3.4) {
    \textbf{Adder}\\[3pt]
    \scriptsize (PC + Imm)
  };

  % =========================================================================
  % 4. MEMORY STAGE: PURE DISTRIBUTED RAM & UNALIGNED ENGINE
  % =========================================================================
  \node[block, minimum width=2.8cm, minimum height=3.4cm] (dmem) at (14.8, 0) {
    \textbf{Data Memory Unit}\\[4pt]
    \scriptsize 4 Banked RAMs ($4 \times 512 \times 8$)\\[4pt]
    \scriptsize Unaligned Barrel Shifter\\[4pt]
    \scriptsize Write-Spy Debug Registers\\[8pt]
    \raggedright
    \tiny \hspace{4pt}Address \hfill ReadData \hspace{4pt}\\[6pt]
    \tiny \hspace{4pt}WriteData\\[6pt]
    \tiny \hspace{4pt}MemRead / MemWrite
  };

  % =========================================================================
  % 5. WRITEBACK STAGE: 4-WAY WRITEBACK MUX
  % =========================================================================
  \node[mux, rotate=90, minimum width=2.6cm, minimum height=0.8cm] (wbmux) at (17.5, 0) {};
  \node at (17.5, 0.9) {\tiny 0};
  \node at (17.5, 0.3) {\tiny 1};
  \node at (17.5, -0.3) {\tiny 2};
  \node at (17.5, -0.9) {\tiny 3};
  \node[above=2pt of wbmux, font=\tiny] {WBSel};

  % Next-PC Mux
  \node[mux, rotate=90, minimum width=1.8cm, minimum height=0.7cm] (pcmux) at (-1.8, 0) {};
  \node at (-1.8, 0.4) {\tiny 0};
  \node at (-1.8, -0.4) {\tiny 1};
  \node[above=2pt of pcmux, font=\tiny] {PCSel};

  % =========================================================================
  % WIRING & INTERCONNECTIONS
  % =========================================================================

  % PC to IMEM
  \draw[bus] (pc.east) -- (imem.west) node[midway, above, font=\tiny] {Addr [10:2]};
  \node[dot] at (0.8, 0) {};
  \draw[bus] (0.8, 0) |- (add_pc4.west);
  \draw[bus] (0.8, 0) |- ++(0, 4.4) -| (add_branch.west);

  % Add PC+4
  \draw[bus] (add_pc4.east) -| (pcmux.south) node[pos=0.2, above, font=\tiny] {PC + 4};
  \node[dot] at (3.0, 3.4) {};
  \draw[bus] (3.0, 3.4) -- ++(0, 2.8) -- (17.5, 6.2) -- (wbmux.north); % to WB Mux input 2

  % IMEM output bus (Instruction [31:0])
  \draw[bus] (imem.east) -- ++(0.7, 0) coordinate (instr_split);
  \node[dot] at (instr_split) {};
  \node[above, font=\tiny] at (instr_split) {Instr [31:0]};

  % Split instruction to:
  % 1. Control Unit: Opcode [6:0], funct3 [14:12], funct7 [31:25]
  \draw[bus] (instr_split) |- (ctrl.west) node[pos=0.7, above, font=\tiny] {Opcode, funct3, funct7};

  % 2. Register File: rs1 [19:15], rs2 [24:20], rd [11:7]
  \draw[bus] (instr_split) -- ++(1.4, 0) |- ([yshift=14pt]rf.west) node[pos=0.8, above, font=\tiny] {rs1 [19:15]};
  \draw[bus] (instr_split) -- ++(1.4, 0) |- ([yshift=0pt]rf.west) node[pos=0.8, above, font=\tiny] {rs2 [24:20]};
  \draw[bus] (instr_split) -- ++(1.4, 0) |- ([yshift=-14pt]rf.west) node[pos=0.8, above, font=\tiny] {rd [11:7]};

  % 3. Immediate Generator
  \draw[bus] (instr_split) |- (imm.west) node[pos=0.8, above, font=\tiny] {Instr [31:7]};

  % RF Outputs:
  % rdata1 -> ALU Input A
  \draw[bus] ([yshift=16pt]rf.east) -- ([yshift=16pt]alu.west) node[pos=0.5, above, font=\tiny] {rdata1 [31:0]};

  % rdata2 -> ALUSrc Mux input 0 & DataMem WriteData
  \draw[bus] ([yshift=-12pt]rf.east) -- ++(0.6, 0) coordinate (rdata2_split);
  \node[dot] at (rdata2_split) {};
  \draw[bus] (rdata2_split) -- (alusrc.south) node[pos=0.5, above, font=\tiny] {rdata2};
  \draw[bus] (rdata2_split) |- ++(0, -2.4) -- (13.2, -2.8) |- ([yshift=-14pt]dmem.west) node[pos=0.8, above, font=\tiny] {WriteData [31:0]};

  % ImmGen Output -> ALUSrc Mux input 1 & Branch Adder
  \draw[bus] (imm.east) -- ++(0.8, 0) coordinate (imm_split);
  \node[dot] at (imm_split) {};
  \draw[bus] (imm_split) |- (alusrc.north) node[pos=0.7, below, font=\tiny] {ImmExt [31:0]};
  \draw[bus] (imm_split) -- ++(1.6, 0) |- (add_branch.west) node[pos=0.85, above, font=\tiny] {ImmExt};

  % ALUSrc Mux -> ALU Input B
  \draw[bus] (alusrc.east) -- ([yshift=-16pt]alu.west) node[pos=0.5, above, font=\tiny] {Operand B};

  % ALU Result -> DataMem Addr & WB Mux input 0
  \draw[bus] ([yshift=10pt]alu.east) -- ++(0.7, 0) coordinate (alu_res_split);
  \node[dot] at (alu_res_split) {};
  \draw[bus] (alu_res_split) -- ([yshift=10pt]dmem.west) node[pos=0.5, above, font=\tiny] {ALU Result (Addr)};
  \draw[bus] (alu_res_split) |- ++(0, 1.8) -- (16.8, 2.2) |- ([yshift=16pt]wbmux.south); % input 0

  % Data Memory ReadData -> WB Mux input 1
  \draw[bus] ([yshift=10pt]dmem.east) -- ([yshift=5pt]wbmux.south) node[pos=0.5, above, font=\tiny] {ReadData [31:0]};

  % ImmGen to WB Mux input 3 (LUI)
  \draw[bus] (imm_split) |- ++(0, -1.0) -- (16.8, -4.2) |- ([yshift=-16pt]wbmux.south) node[pos=0.85, below, font=\tiny] {ImmExt (LUI)};

  % WB Mux Output -> RegFile WriteData
  \draw[bus] (wbmux.east) -- ++(0.5, 0) -- ++(0, -4.8) -- (5.0, -4.8) |- ([yshift=-28pt]rf.west) node[pos=0.85, above, font=\tiny] {WriteData [31:0]};

  % Branch Target Adder Output -> PC Mux input 1
  \draw[bus] (add_branch.east) -- ++(0.5, 0) -- ++(0, 2.2) -- (-2.4, 5.6) |- ([yshift=-8pt]pcmux.south) node[pos=0.9, above, font=\tiny] {Branch Target};

  % PC Mux -> PC Register
  \draw[bus] (pcmux.east) -- (pc.west) node[midway, above, font=\tiny] {Next PC};

  % =========================================================================
  % CONTROL LINES (DASHED)
  % =========================================================================
  % Ctrl -> RegWrite
  \draw[ctrlwire] ([xshift=-10pt]ctrl.south) -- ([xshift=-10pt, yshift=28pt]rf.north) node[midway, right, font=\tiny] {RegWrite};

  % Ctrl -> ALUSrc
  \draw[ctrlwire] ([xshift=15pt]ctrl.south) |- (alusrc.west) node[pos=0.7, right, font=\tiny] {ALUSrc};

  % Ctrl -> ALUOp
  \draw[ctrlwire] ([xshift=35pt]ctrl.south) |- ([yshift=20pt]alu.north) node[pos=0.7, right, font=\tiny] {ALUOp [3:0]};

  % Ctrl -> MemRead / MemWrite
  \draw[ctrlwire] (ctrl.east) -| ([xshift=-10pt]dmem.north) node[pos=0.6, above, font=\tiny] {MemRead / MemWrite};

  % Ctrl -> WBSel
  \draw[ctrlwire] (ctrl.east) -| (wbmux.west) node[pos=0.8, above, font=\tiny] {WBSel [1:0]};

  % Branch evaluation to PCSel logic
  \node[block, draw=black, minimum width=1.1cm, minimum height=0.9cm] (br_and) at (10.0, 4.4) {\tiny AND\\[-2pt]\tiny Gate};
  \draw[ctrlwire] ([xshift=10pt]ctrl.east) -- (br_and.west) node[midway, above, font=\tiny] {Branch};
  \draw[ctrlwire] ([yshift=-12pt]alu.east) -| (br_and.south) node[pos=0.3, right, font=\tiny] {Zero / Cond};
  \draw[ctrlwire] (br_and.north) |- ++(0, 0.8) -- (-1.8, 5.2) -- (pcmux.west) node[pos=0.9, left, font=\tiny] {PCSel};

\end{tikzpicture}
\end{document}
